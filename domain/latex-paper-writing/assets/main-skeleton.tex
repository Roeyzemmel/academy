  \uppercase{#1}%

  \expandafter\newcommand\csname #1\endcsname[1]{% Defines \Firstname{...comment...}

      \color{#3}(\firsttwoletters#2\@nil\theCOMMENTS)%
    }%

      {\color{#3}(\expandafter\uppercase\expandafter{\firsttwoletters#2\@nil}\theCOMMENTS)#1: 
      }} ##1}%
    \stepcounter{COMMENTS}%
  }%

}
